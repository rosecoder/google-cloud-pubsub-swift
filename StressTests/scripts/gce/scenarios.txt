main s1-100subs subs=100 rate=10000 duration=900 batch=10 publishers=64 max-outstanding=200 max-outstanding-bytes=20971520 sleep-ms=20 sleep-jitter=0.5 sample=10

main s2c-ceiling100 subs=100 rate-steps=10000:60,20000:60,40000:60,60000:60 publishers=128 batch=100 max-backlog=200000 max-outstanding=200 max-outstanding-bytes=20971520 drain=300 sample=5

main s6-shutdown mode=churn subs=50 rate=20000 cycles=2 cycle-min=60 cycle-max=60 sleep-ms=200 sleep-jitter=0.5 shutdown-timeout=10 max-outstanding=200 max-outstanding-bytes=20971520 publishers=128 batch=100 max-backlog=200000 drain=300 sample=5
